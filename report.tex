	\section{Описание задачи и данных}
	
	\subsection{Постановка задачи}
	
	Задача --- предсказание \textbf{Customer Lifetime Value (LTV)} на горизонте 30 дней.
	Модель должна по истории активности пользователя предсказать суммарный GMV
	за период с 14.02.2026 по 15.03.2026.
	
	\begin{dangerbox}
		\textbf{Публичный / Приватный лидерборд: 20\% / 80\%!}\\
		Оценка на публичном лидерборде вычисляется только по 50\,000 из 250\,000 клиентов.
		Это означает высокий риск \emph{public-private shake-up}.
	\end{dangerbox}
	
	\textbf{Метрика качества:}
	\[
	\mathrm{RMSLE} = \sqrt{\frac{1}{n}\sum_{i=1}^{n}\bigl(\log(1+\hat{y}_i) - \log(1+y_i)\bigr)^2}
	\]
	
	\subsection{Особенности исходных данных}
	
	\begin{infobox}
		\textbf{Объём:} 30+ млн записей, 250\,000 клиентов\\
		\textbf{Период:} 01.01.2025 -- 13.02.2026 (407 дней)\\
		\textbf{Тип:} Прореженные временные ряды (спарс).
	\end{infobox}
	
	Особенности целевой переменной: данные сильно \textbf{zero-inflated}. Около 45.1\% пользователей в обучающей выборке не совершают ни одной покупки (target = 0).

	\section{Схема валидации: Walk-Forward Cross-Validation}
	
	Для временных рядов используется строго \textbf{Walk-Forward CV}.
	Тестировались два подхода к формированию обучающей выборки:
	\begin{enumerate}
		\item \textbf{Single-Fold:} обучение только на предыдущем фолде ($\mathcal{D}_{val-1} \to \mathcal{D}_{val}$). Подход показал наилучшую устойчивость.
		\item \textbf{Cumulative:} обучение на всех исторических фолдах ($\bigcup_{j=0}^{val-1} \mathcal{D}_j \to \mathcal{D}_{val}$).
	\end{enumerate}
	
	Параметр \texttt{stride} (шаг смещения anchor date) критически влияет на качество. При $stride=7$ и $target=30$ возникало 77\%-е перекрытие таргетов между фолдами, что приводило к сильному переобучению алгоритмов (в частности, Optuna). Оптимальным решением признан $stride=14$.

	\section{Архитектура базовых моделей}
	
	\subsection{CatBoostRegressor (Основа решения)}
	Градиентный бустинг с \emph{level-wise} построением деревьев. Обучение производилось по схеме \textbf{Single-Fold} на 392 признаках v4. Выдал лучший одиночный результат \rmsleGood{1.66049}.
	
	\subsection{LightGBM}
	Градиентный бустинг с \emph{leaf-wise} сплитом. Настроен через Optuna. Благодаря другой архитектуре построения деревьев, модель формирует иное распределение ошибок, что отлично подходит для диверсификации.
	
	\subsection{Рекуррентная нейронная сеть (PyTorch GRU)}
	\textbf{RNN учитывает секвенциальную природу данных}, оперируя окнами в 180 дней. Архитектура: 3-слойный GRU (hidden\_dim=192), Optimizer AdamW + CosineAnnealingLR.
	
	\begin{successbox}
		\textbf{Роль RNN в пайплайне:}\\
		Именно RNN обеспечила необходимую \textbf{ортогональность ошибок}. Смешивание 90\% предсказаний CatBoost и 10\% предсказаний RNN дало синергетический эффект, установив абсолютный рекорд на лидерборде.
	\end{successbox}

	\section{Анализ неудачных подходов (BTYD и Zero-Hack)}
	
	\subsection{BTYD (BG/NBD + Gamma-Gamma)}
	Вероятностная модель сильно завысила ожидания (средний предсказанный LTV = 85.17, против $\approx 35$ у CatBoost). Отправка чистого сабмита дала скор $>1.81$.
	
	\subsection{Провал метода «Zero-Hack»}
	В модели Two-Stage классификатор предсказывал вероятность покупки $P(buy)$. Была выдвинута гипотеза занулять финальный GMV для пользователей с $P(buy) < 0.46$. На публичном лидерборде этот метод показал \textbf{катастрофическое падение скора (1.73573)}. 
	\textbf{Вывод:} Метрика RMSLE крайне жестко штрафует за предсказание абсолютного нуля, если клиент всё же совершил покупку. Искусственное зануление признано опасным и исключено из пайплайна.

	\section{Стратегия на финальные отправки}
	
	Текущий рекорд 1.65935 достигнут за счёт ансамблирования. Дальнейшие шаги фокусируются на эксплуатации этой синергии:
	
	\begin{enumerate}
		\item \textbf{Тонкий тюнинг весов (Micro-blending):} Так как пропорция 90/10 превзошла 85/15, необходимо исследовать крайние значения (например, 95/5 или 93/7), чтобы найти абсолютный оптимум.
		\item \textbf{Перезапуск LightGBM:} Обучение LightGBM на корректном топе признаков v4.
		\item \textbf{Разработка RNN v4:} Обучение нейросети на обновленных данных (использование \texttt{SEQ\_LEN = 180} и включение флагов каталога/поиска), чтобы повысить качество базового предикта перед блендингом.
		\item \textbf{Pseudo-labeling (Псевдо-разметка):} Использование наиболее уверенных предсказаний из текущего лучшего сабмита в качестве истинных меток для расширения обучающей выборки CatBoost. Этот метод позволяет модели выучить скрытую структуру тестовой выборки.
	\end{enumerate}

	\subsection*{Оценка качества (RMSLE)}
	\[
	\mathrm{RMSLE} = \sqrt{\frac{1}{n}\sum_{i=1}^{n}\bigl(\log(1+\hat{y}_i) - \log(1+y_i)\bigr)^2}
	\]
	
	\subsection*{Log-трансформация}
	\[
	\tilde{y} = \log\!\bigl(1 + \max(y,\,0)\bigr),\qquad
	\hat{y} = \max\!\bigl(e^{\hat{\tilde{y}}} - 1,\;0\bigr)
	\]
